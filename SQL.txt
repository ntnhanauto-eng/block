CREATE TABLE IF NOT EXISTS `rooms` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `room_name` varchar(50) NOT NULL,
  `status` enum('trong','khach','ve_sink') NOT NULL DEFAULT 'trong',
  `tuya_device_id` varchar(100) DEFAULT NULL, -- Điền ID cảm biến gắn với phòng này
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `room_logs` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `room_id` int(11) NOT NULL,
  `event_type` varchar(50) NOT NULL, -- THƯỜNG hoặc BẤT THƯỜNG
  `details` varchar(255) NOT NULL,   -- Ví dụ: Cửa mở lúc khách vắng mặt
  `event_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
