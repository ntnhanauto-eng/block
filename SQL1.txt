-- 1. Bảng lưu tài khoản (Mật khẩu được mã hóa chuẩn bằng password_hash)
CREATE TABLE IF NOT EXISTS `users` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `username` VARCHAR(50) NOT NULL UNIQUE,
  `password` VARCHAR(255) NOT NULL,
  `role` ENUM('admin', 'staff') DEFAULT 'staff'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `rooms` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `room_name` VARCHAR(50) NOT NULL,
  `device_id` VARCHAR(50) NOT NULL,
  `status` ENUM('trong', 'khach', 've_sinh') DEFAULT 'trong'

-- 3. Bảng lưu lịch sử đóng mở cửa cảm biến
CREATE TABLE IF NOT EXISTS `door_logs` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `room_id` INT,
  `event_time` DATETIME DEFAULT CURRENT_TIMESTAMP,
  `event_type` VARCHAR(20), -- 'Bình thường' hoặc 'BẤT THƯỜNG'
  `details` TEXT,
  FOREIGN KEY (`room_id`) REFERENCES `rooms`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
